    function handlePhotoUpload(e) {
      const files = e.target.files;
      if(files.length + tempAssetPhotos.length > 5) {
        Swal.fire({ icon: 'warning', title: 'Batas Upload', text: 'Maksimal 5 foto per aset.', background: '#1e293b', color: '#fff' });
        return;
      }
      Array.from(files).forEach(file => {
        const reader = new FileReader();
        reader.onload = function(ev) {
          tempAssetPhotos.push(ev.target.result);
          renderPhotoPreviews();
        };
        reader.readAsDataURL(file);
      });
    }

    function deleteRecord(sheetKey, colIdx, val) {
      Swal.fire({
        title: 'Hapus Data?',
        text: 'Apakah Anda yakin ingin menghapus record ini?',
        icon: 'warning',
        showCancelButton: true,
        confirmButtonColor: '#ef4444',
        cancelButtonColor: '#334155',
        confirmButtonText: 'Ya, Hapus',
        cancelButtonText: 'Batal',
        background: '#1e293b',
        color: '#fff'
      }).then(res => {
        if(res.isConfirmed) {
          if(typeof google !== 'undefined' && google.script && google.script.run) {
            google.script.run.withSuccessHandler(() => { loadAllDataFromBackend(); }).deleteRecord(sheetKey, colIdx, val);
          } else {
            appData[sheetKey] = appData[sheetKey].filter(item => String(Object.values(item)[colIdx]) !== String(val));
            renderAll();
          }
        }
      });
    }
